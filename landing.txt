    // --- 6. 3D MAP ENGINE (Vanilla Canvas) ---
    function initMapEngine() {
      const canvas = document.getElementById('engine-canvas');
      const ctx = canvas.getContext('2d');
      const wrapper = document.getElementById('engine-wrapper');
      
      let cam = { rx: 0.2, ry: 0.3, z: 1000 };
      let tcam = { rx: 0.2, ry: 0.3, z: 1000 };
      let dragging = false, lx = 0, ly = 0, velX = 0, velY = 0;
      let time = 0;
      let isPaused = false;

      function resize() {
        canvas.width = wrapper.offsetWidth;
        canvas.height = wrapper.offsetHeight;
      }
      window.addEventListener('resize', resize);
      resize();

      // Pre-render blobs to images for performance in canvas
      const blobImages = {};
      ARCHETYPES.forEach(arch => {
        const svgStr = getBlobSVG(arch, 150);
        const img = new Image();
        // Fixed the way we encode the SVG to avoid 'broken' state errors
        img.src = 'data:image/svg+xml;base64,' + btoa(unescape(encodeURIComponent(svgStr)));
        blobImages[arch.id] = img;
      });

      function project(x, y, z) {
        const cX = Math.cos(cam.rx), sX = Math.sin(cam.rx);
        const y1 = y * cX - z * sX, z1 = y * sX + z * cX;
        const cY = Math.cos(cam.ry), sY = Math.sin(cam.ry);
        const x2 = x * cY + z1 * sY, z2 = -x * sY + z1 * cY;
        const depth = 1000 + z2 + cam.z;
        if (depth < 1) return null;
        const sc = 1000 / depth;
        return { sx: canvas.width / 2 + x2 * sc, sy: canvas.height / 2 + y1 * sc, sc, z2 };
      }

      function loop() {
        if(!isPaused) time += 0.01;
        
        if (!dragging && !isPaused) {
          tcam.ry += 0.002;
          tcam.rx += velX; tcam.ry += velY;
          velX *= 0.95; velY *= 0.95;
        }
        
        tcam.rx = Math.max(-0.8, Math.min(0.8, tcam.rx));
        cam.rx += (tcam.rx - cam.rx) * 0.1;
        cam.ry += (tcam.ry - cam.ry) * 0.1;

        ctx.clearRect(0, 0, canvas.width, canvas.height);

        // Nodes projection
        const proj = ARCHETYPES.map((arch, i) => {
          const fY = Math.sin(time + i * 0.9) * 15;
          const p = project(arch.pos.x, arch.pos.y + fY, arch.pos.z);
          return p ? { arch, p } : null;
        }).filter(Boolean);
        
        proj.sort((a, b) => b.p.z2 - a.p.z2);
        const projMap = Object.fromEntries(proj.map(n => [n.arch.id, n.p]));

        // Edges
        EDGES.forEach(([i, j]) => {
          const aId = ARCHETYPES[i].id, bId = ARCHETYPES[j].id;
          const pa = projMap[aId], pb = projMap[bId];
          if (!pa || !pb) return;
          ctx.beginPath();
          ctx.setLineDash([6, 8]);
          ctx.strokeStyle = `rgba(28,25,23,0.2)`;
          ctx.lineWidth = 2;
          const mx = (pa.sx + pb.sx)/2 - (pa.sy - pb.sy)*0.1;
          const my = (pa.sy + pb.sy)/2 + (pa.sx - pb.sx)*0.1;
          ctx.moveTo(pa.sx, pa.sy);
          ctx.quadraticCurveTo(mx, my, pb.sx, pb.sy);
          ctx.stroke();
        });

        // Draw Nodes
        proj.forEach(({ arch, p }) => {
          const sz = 120 * p.sc;
          if (sz < 10) return;
          
          // Glow
          const g = ctx.createRadialGradient(p.sx, p.sy, 0, p.sx, p.sy, sz*0.8);
          g.addColorStop(0, arch.color + '66'); g.addColorStop(1, arch.color + '00');
          ctx.beginPath(); ctx.arc(p.sx, p.sy, sz*0.8, 0, Math.PI*2);
          ctx.fillStyle = g; ctx.fill();

          // Check if the image is loaded AND its width is not 0 (which would indicate a broken state even if complete is true)
          if (blobImages[arch.id] && blobImages[arch.id].complete && blobImages[arch.id].naturalWidth !== 0) {
            ctx.globalAlpha = Math.min(1, p.sc * 1.5);
            ctx.drawImage(blobImages[arch.id], p.sx - sz/2, p.sy - sz/2, sz, sz);
            ctx.globalAlpha = 1;
          }
        });

        requestAnimationFrame(loop);
      }
      
      // Controls
      wrapper.addEventListener('pointerdown', e => { dragging = true; lx = e.clientX; ly = e.clientY; velX=0; velY=0; });
      window.addEventListener('pointermove', e => {
        if(!dragging) return;
        const dx = e.clientX - lx, dy = e.clientY - ly;
        tcam.ry += dx * 0.005; tcam.rx += dy * 0.005;
        velY = dx * 0.0005; velX = dy * 0.0005;
        lx = e.clientX; ly = e.clientY;
      });
      window.addEventListener('pointerup', () => dragging = false);

      document.getElementById('engine-pause').onclick = function() {
        isPaused = !isPaused;
        this.innerText = isPaused ? 'Play' : 'Pause';
      };

      loop();
    }

    // --- 7. GENERAL UI INTERACTIVITY ---
    function initUI() {
      // Scroll Reveal
      const observer = new IntersectionObserver((entries) => {
        entries.forEach(e => {
          if (e.isIntersecting) { e.target.classList.add('is-revealed'); observer.unobserve(e.target); }
        });
      }, { threshold: 0.1 });
      document.querySelectorAll('.reveal-on-scroll').forEach(el => observer.observe(el));

      // Navbar
      window.addEventListener('scroll', () => {
        const nav = document.getElementById('nav-container');
        if (window.scrollY > 50) {
          nav.classList.add('bg-paper/95', 'backdrop-blur-xl', 'py-1');
          nav.classList.remove('bg-paper/70', 'py-2.5', 'md:py-3');
        } else {
          nav.classList.add('bg-paper/70', 'py-2.5', 'md:py-3');
          nav.classList.remove('bg-paper/95', 'backdrop-blur-xl', 'py-1');
        }
      });

      // Mobile Menu
      const menu = document.getElementById('mobile-menu');
      document.getElementById('mobile-menu-btn').onclick = () => {
        menu.classList.remove('opacity-0', 'invisible', 'pointer-events-none');
        menu.classList.add('opacity-100', 'visible');
      };
      document.getElementById('close-menu-btn').onclick = () => {
        menu.classList.add('opacity-0', 'invisible', 'pointer-events-none');
        menu.classList.remove('opacity-100', 'visible');
      };
      menu.querySelectorAll('a').forEach(a => a.onclick = () => document.getElementById('close-menu-btn').click());

      // Horizontal Drag for Codex
      const codex = document.getElementById('codex-container');
      let isDown = false, startX, scrollLeft;
      codex.addEventListener('mousedown', e => { isDown = true; startX = e.pageX - codex.offsetLeft; scrollLeft = codex.scrollLeft; });
      codex.addEventListener('mouseleave', () => isDown = false);
      codex.addEventListener('mouseup', () => isDown = false);
      codex.addEventListener('mousemove', e => {
        if (!isDown) return;
        e.preventDefault();
        const x = e.pageX - codex.offsetLeft;
        codex.scrollLeft = scrollLeft - (x - startX);
      });
    }

      initSynergy();
      initMapEngine();
      initUI();
    };
