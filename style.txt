/** 4D SPATIAL ENGINE */
const viewport = document.getElementById('center-viewport');

function initEngine() {
    // Environmental Props (Islands / Clouds)
    const envProps = [
        { type: 'island', x: 0, y: 150, z: 0, s: 1.5 },
        { type: 'cloud', x: -400, y: -200, z: -300, s: 1 },
        { type: 'cloud', x: 500, y: 100, z: 200, s: 1.2 },
        { type: 'cloud', x: 0, y: -400, z: 500, s: 2 }
    ];
    envProps.forEach((p, i) => {
        const el = document.createElement('div');
        el.className = 'env-obj'; el.id = `env-${i}`;
        el.innerHTML = p.type === 'island' ? getIslandSVG() : getCloudSVG();
        el.dataset.x = p.x; el.dataset.y = p.y; el.dataset.z = p.z; el.dataset.scale = p.s;
        if(p.type === 'island') el.querySelector('svg').classList.add('island-svg');
        else el.querySelector('svg').classList.add('cloud-svg');
        engine3d.appendChild(el);
    });

    // Nodes
    DATA.forEach(n => {
        const el = document.createElement('div');
        el.className = 'node-wrapper'; el.id = `node-${n.id}`;
        el.innerHTML = `<div class="node-highlight"></div><div class="blob-char">${getBlobSVG(n.color, n.mood)}</div><div class="node-label">${n.name}</div>`;
        
        el.addEventListener('pointerdown', e => { e.stopPropagation(); selectNode(n.id); });
        nodesLayer.appendChild(el);
        nodeEls[n.id] = el;
    });

    // Edges
    EDGES.forEach(([a, b]) => {
        const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
        path.setAttribute('class', 'link-line');
        edgesSvg.appendChild(path);
        edgeEls.push({ s: a, t: b, el: path });
    });
}

function proj3D(x, y, z) {
    let cX = Math.cos(cam.rx), sX = Math.sin(cam.rx);
    let y1 = y * cX - z * sX, z1 = y * sX + z * cX;
    
    let cY = Math.cos(cam.ry), sY = Math.sin(cam.ry);
    let x2 = x * cY + z1 * sY, z2 = -x * sY + z1 * cY;
    
    let scale = fov / (fov + z2 + cam.z);
    return { x: x2*scale, y: y1*scale, s: Math.max(0.01, scale), zD: z2 };
}

function render() {
    if(!isDrag) {
        if(autoSpin) tCam.ry += 0.001;
        tCam.rx += vel.rx; tCam.ry += vel.ry;
        vel.rx *= 0.9; vel.ry *= 0.9;
    }
    cam.rx += (tCam.rx - cam.rx)*0.1; cam.ry += (tCam.ry - cam.ry)*0.1; cam.z += (tCam.z - cam.z)*0.1;
    cam.rx = Math.max(-0.8, Math.min(0.8, cam.rx)); // Clamp pitch

    let projMap = {};
    let vw = viewport.clientWidth/2, vh = viewport.clientHeight/2;

    // Render Env
    document.querySelectorAll('.env-obj').forEach(el => {
        const p = proj3D(parseFloat(el.dataset.x), parseFloat(el.dataset.y), parseFloat(el.dataset.z));
        if(p.s <= 0.01) { el.style.display = 'none'; return; }
        el.style.display = 'block';
        const baseS = parseFloat(el.dataset.scale);
        el.style.transform = `translate3d(${p.x}px, ${p.y}px, 0) scale(${p.s * baseS})`;
        el.style.zIndex = Math.round(p.s * 100);
        el.style.filter = `blur(${p.zD > 200 ? (p.zD-200)*0.01 : 0}px)`;
    });

    // Render Nodes
    DATA.forEach(n => {
        const p = proj3D(n.x, n.y, n.z);
        projMap[n.id] = p;
        const el = nodeEls[n.id];
        if(p.s <= 0.01) { el.style.display = 'none'; return; }
        el.style.display = 'flex';
        
        el.style.transform = `translate(-50%, -50%) translate3d(${p.x}px, ${p.y}px, 0) scale(${p.s})`;
        el.style.zIndex = Math.round(p.s * 1000);
        
        const dim = el.classList.contains('dimmed');
        el.style.filter = `blur(${p.zD > 200 ? (p.zD-200)*0.02 : 0}px) ${dim ? 'grayscale(80%)' : ''}`;
        el.style.opacity = dim ? 0.3 : 1;
    });

    // Render Edges
    edgeEls.forEach(edge => {
        const pA = projMap[edge.s], pB = projMap[edge.t];
        if(pA.s<=0.01 || pB.s<=0.01) { edge.el.style.display = 'none'; return; }
        edge.el.style.display = 'block';
        
        const x1 = vw + pA.x, y1 = vh + pA.y, x2 = vw + pB.x, y2 = vh + pB.y;
        
        // Curved trajectory
        const mx = (x1+x2)/2 - (y2-y1)*0.1, my = (y1+y2)/2 + (x2-x1)*0.1;
        edge.el.setAttribute('d', `M ${x1} ${y1} Q ${mx} ${my} ${x2} ${y2}`);
        
        if(activeId) {
            if(edge.s === activeId || edge.t === activeId) { edge.el.style.opacity = 0.8; edge.el.style.stroke = 'var(--c-orange)'; }
            else { edge.el.style.opacity = 0.1; edge.el.style.stroke = 'var(--ink-main)'; }
        } else {
            edge.el.style.opacity = 0.4; edge.el.style.stroke = 'var(--ink-main)';
        }
    });

    requestAnimationFrame(render);
}

/** INTERACTION */
let activeId = null;

function selectNode(id) {
    if(activeId === id) return;
    activeId = id;
    const nd = DATA.find(n => n.id === id);

    Object.values(nodeEls).forEach(el => el.classList.add('dimmed'));
    document.querySelectorAll('.node-wrapper').forEach(el => el.classList.remove('active'));
    
    nodeEls[id].classList.remove('dimmed');
    nodeEls[id].classList.add('active');

    const conn = new Set();
    EDGES.forEach(([a, b]) => { if(a===id) conn.add(b); if(b===id) conn.add(a); });
    conn.forEach(cid => nodeEls[cid].classList.remove('dimmed'));

    // Populate Inspector
    const rp = document.getElementById('right-panel-container');
    rp.classList.remove('hidden');
    if(window.innerWidth <= 900) {
        document.getElementById('overlay').classList.add('active');
    }

    document.getElementById('insp-title').textContent = nd.name;
    document.getElementById('insp-sub').textContent = nd.sub;
    document.getElementById('insp-desc').textContent = nd.desc;
    document.getElementById('insp-blob').innerHTML = getBlobSVG(nd.color, nd.mood);

    // Animate Bars
    setTimeout(() => {
        document.getElementById('bar-agency').style.width = `${nd.agency}%`;
        document.getElementById('bar-temp').style.width = `${nd.temp}%`;
        document.getElementById('bar-focus').style.width = `${nd.focus}%`;
        document.getElementById('bar-attach').style.width = `${nd.attach}%`;
    }, 50);

    // Related
    const rg = document.getElementById('insp-related');
    rg.innerHTML = '';
    Array.from(conn).slice(0,3).forEach(cid => {
        const rn = DATA.find(n => n.id === cid);
        const cd = document.createElement('div'); cd.className = 'rcard';
        cd.onclick = (e) => { e.stopPropagation(); selectNode(cid); };
        cd.innerHTML = `${getBlobSVG(rn.color, rn.mood)}<span>${rn.name}</span>`;
        rg.appendChild(cd);
    });

    // Flight Camera
    autoSpin = false;
    tCam.ry = -Math.atan2(nd.x, nd.z + 1000);
    tCam.rx = Math.atan2(nd.y, Math.sqrt(nd.x*nd.x + Math.pow(nd.z+1000, 2)));
    tCam.z = 800; // zoom in
}

    activeId = null;
    document.getElementById('right-panel-container').classList.add('hidden');
    Object.values(nodeEls).forEach(el => el.classList.remove('active', 'dimmed'));
    document.getElementById('overlay').classList.remove('active');
    
    // reset bars
    document.querySelectorAll('.attr-fill').forEach(b => b.style.width = '0%');
}

});

document.querySelectorAll('.track').forEach(tr => {
    let slide = false;
    tr.addEventListener('pointerdown', e => { slide=true; doSlide(e, tr); });
    window.addEventListener('pointermove', e => { if(slide) doSlide(e, tr); });
    window.addEventListener('pointerup', () => slide=false);
});
function doSlide(e, tr) {
    const r = tr.getBoundingClientRect();
    let p = Math.max(0, Math.min(1, (e.clientX - r.left)/r.width));
    tr.querySelector('.fill').style.width = `${p*100}%`;
    tr.querySelector('.thumb').style.left = `${p*100}%`;
}

const dust = Array.from({length:60}, ()=>({ x:Math.random()*2000, y:Math.random()*2000, r:Math.random()*1.5+0.5, a:Math.random() }));
function animDust() {
    dCvs.width = viewport.clientWidth; dCvs.height = viewport.clientHeight;
    dCtx.fillStyle='#fff';
    dust.forEach(d => {
        d.y -= 0.5; d.a += 0.05;
        if(d.y < 0) { d.y = dCvs.height; d.x = Math.random()*dCvs.width; }
        dCtx.globalAlpha = 0.2 + Math.sin(d.a)*0.3;
        dCtx.beginPath(); dCtx.arc(d.x % dCvs.width, d.y, d.r, 0, 7); dCtx.fill();
    });
    requestAnimationFrame(animDust);
}
